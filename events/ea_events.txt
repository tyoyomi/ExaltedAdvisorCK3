# Management decision event — advisors
ea_events.2001 = {
	type = character_event
	title = ea_events.2001.t
	desc = ea_events.2001.desc
	theme = diplomacy

	immediate = {
		if = {
			limit = {
				any_in_list = {
					variable = ea_bound_advisors
					count >= 1
				}
			}
			every_in_list = {
				variable = ea_bound_advisors
				limit = {
					is_alive = yes
					has_trait = ea_exalted_advisor
					exists = var:ea_patron
					var:ea_patron = root
				}
				add_to_list = ea_my_advisors
			}
		}
		else = {
			every_vassal = {
				limit = {
					has_trait = ea_exalted_advisor
					exists = var:ea_patron
					var:ea_patron = root
				}
				add_to_list = ea_my_advisors
			}
			every_courtier_or_guest = {
				limit = {
					has_trait = ea_exalted_advisor
					exists = var:ea_patron
					var:ea_patron = root
				}
				add_to_list = ea_my_advisors
			}
		}
	}

	option = {
		name = ea_events.2001.a
		custom_tooltip = ea_events.2001.a_tt
	}

	option = {
		name = ea_events.2001.b
		trigger = {
			any_in_list = {
				list = ea_my_advisors
				count >= 1
			}
		}
		custom_tooltip = ea_events.2001.b_tt
		# Point players at the revoke interaction rather than duplicating logic
	}

	option = {
		name = ea_events.2001.c
		custom_tooltip = ea_events.2001.c_tt
		ea_auto_assign_exalted_advisors_for_patron_effect = yes
		trigger_event = ea_events.1501
	}
}
