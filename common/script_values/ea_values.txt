ea_exalted_advisor_count_for_patron_value = {
	value = 0
	if = {
		limit = {
			any_in_list = {
				variable = ea_bound_advisors
				count >= 1
			}
		}
		every_in_list = {
			variable = ea_bound_advisors
			limit = {
				is_alive = yes
				has_trait = ea_exalted_advisor
				exists = var:ea_patron
				var:ea_patron = root
			}
			add = 1
		}
	}
	else = {
		every_vassal = {
			limit = {
				has_trait = ea_exalted_advisor
				exists = var:ea_patron
				var:ea_patron = root
			}
			add = 1
		}
		every_courtier_or_guest = {
			limit = {
				has_trait = ea_exalted_advisor
				exists = var:ea_patron
				var:ea_patron = root
			}
			add = 1
		}
	}
}

# In-realm serving advisors only (abroad / pending-grace do not count toward aura)
# List-first (O(bound)); realm fallback only when binder list empty (mid-save heal).
ea_serving_advisor_count_for_patron_value = {
	value = 0
	if = {
		limit = {
			any_in_list = {
				variable = ea_bound_advisors
				count >= 1
			}
		}
		every_in_list = {
			variable = ea_bound_advisors
			limit = {
				is_alive = yes
				has_trait = ea_exalted_advisor
				exists = var:ea_patron
				var:ea_patron = root
				ea_in_patron_realm = yes
				NOT = { ea_in_recall_grace = yes }
			}
			add = 1
		}
	}
	else = {
		every_vassal = {
			limit = {
				has_trait = ea_exalted_advisor
				exists = var:ea_patron
				var:ea_patron = root
				ea_in_patron_realm = yes
				NOT = { ea_in_recall_grace = yes }
			}
			add = 1
		}
		every_courtier_or_guest = {
			limit = {
				has_trait = ea_exalted_advisor
				exists = var:ea_patron
				var:ea_patron = root
				ea_in_patron_realm = yes
				NOT = { ea_in_recall_grace = yes }
			}
			add = 1
		}
	}
}
