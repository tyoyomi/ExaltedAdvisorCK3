ea_bestow_exalted_advisor_interaction = {
	category = interaction_category_friendly
	icon = council
	interface_priority = 60
	common_interaction = yes

	desc = ea_bestow_exalted_advisor_interaction_desc

	auto_accept = yes

	is_shown = {
		NOT = { has_game_rule = ea_rule_granting_disabled }
		scope:actor = {
			is_alive = yes
			ea_actor_may_bestow_by_rule = yes
		}
		scope:recipient = {
			is_alive = yes
			NOT = { has_trait = ea_exalted_advisor }
			# Hide when not bound to actor's realm — prevents shown-but-disabled on
			# random characters. Cap / SS exclusion stay in is_valid for tooltips.
			ea_recipient_in_actor_realm = yes
		}
		OR = {
			AND = {
				has_game_rule = ea_rule_self_grant_yes
				scope:actor = scope:recipient
			}
			scope:actor != scope:recipient
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = ea_tt_actor_may_bestow
				ea_can_actor_bestow = yes
			}
		}
		scope:recipient = {
			custom_tooltip = {
				text = ea_tt_can_receive_exalted_advisor
				ea_can_receive_exalted_advisor = yes
			}
		}
	}

	on_accept = {
		scope:actor = { save_scope_as = ea_patron }
		scope:recipient = {
			ea_bestow_exalted_advisor_effect = yes
		}
	}

	ai_targets = {
		ai_recipients = courtiers
		chance = 0.1
	}
	ai_targets = {
		ai_recipients = vassals
		chance = 0.05
	}
	ai_target_quick_trigger = {
		adult = yes
	}
	ai_frequency = 120

	ai_potential = {
		has_game_rule = ea_rule_granting_player_and_ai
		ea_patron_under_advisor_cap = yes
	}

	ai_will_do = {
		base = 0
		modifier = {
			add = 25
			scope:recipient = {
				is_councillor_of = scope:actor
			}
		}
		modifier = {
			add = 10
			scope:actor = {
				opinion = {
					target = scope:recipient
					value >= 50
				}
			}
		}
		modifier = {
			add = -100
			AND = {
				has_game_rule = ea_rule_mutual_exclusion_block_with_sovereign
				scope:recipient = {
					ea_character_is_supreme_sovereign = yes
				}
			}
		}
		modifier = {
			add = -100
			AND = {
				has_game_rule = ea_rule_advisor_vassal_exclusion_block
				scope:recipient = {
					has_trait = ea_exalted_vassal
				}
			}
		}
	}
}

			custom_tooltip = {
				text = ea_tt_not_abroad
				OR = {
					NOT = { ea_in_patron_realm = yes }
					ea_in_recall_grace = yes
				}
			}
			custom_tooltip = {
				text = ea_tt_in_recall_grace
				OR = {
					ea_in_recall_grace = yes
					has_game_rule = ea_rule_strip_on_leave_off
					has_game_rule = ea_rule_recall_grace_on
				}
			}
			ea_is_recallable = yes

	}

	# Prestige cost reserved (@ea_recall_prestige_cost = 0) — free in v0.2

		chance = 0.2
	}

		ai_recipients = scripted_relations
		chance = 0.15
	}
	ai_frequency = 36

	ai_potential = {
		has_game_rule = ea_rule_granting_player_and_ai
		OR = {
			has_game_rule = ea_rule_recall_grace_on
			has_game_rule = ea_rule_strip_on_leave_off
		}
	}

	ai_will_do = {
		base = 0
		modifier = {
			add = 200
			scope:recipient = {
				ea_in_recall_grace = yes
			}
		}

			add = 50

				NOT = { ea_in_patron_realm = yes }
			}
		}
	}
